PROCEDURE drawHoldPiece(Coordinate screenPosition)
	// Get the piece shape unrotated
	holdPiece = gameState.getHoldPiece()
	shape = holdPiece.getShape(0)
	block = holdPiece.getComposingBlock()

	// Iterate through the blocks that make up the piece
	FOR i = 0 TO LEN(shape)
		FOR j = 0 TO LEN(shape[i])
			IF shape[i][j] == true THEN // Only render a block if it is enabled in the shape 2D array
				// Converts coordinates from the shape array to pixel coordinates
				// and adds the screen positions so that it is drawn in the correct place on the canvas
				mappedX = i * columnWidth + screenPosition.x
				mappedY = j * rowHeight + screenPosition.y
				// Draw square with dimensions rowHeight x columnWidth in the block's colour
				drawRectangle(mappedX, mappedY, mappedX + columnWidth, mappedY + rowHeight, block.colour)
			ENDIF
		NEXT j
	NEXT i
ENDPROCEDURE
